cmake_minimum_required(VERSION 3.24)
project(cv40examples_cpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CV40_FETCH_DEPS "Download pinned OpenCV/contrib and dlib when not installed" ON)
include(FetchContent)
foreach(dep opencv opencv_contrib dlib)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/${dep}-src")
    string(TOUPPER "${dep}" dep_upper)
    set(FETCHCONTENT_SOURCE_DIR_${dep_upper} "${CMAKE_CURRENT_SOURCE_DIR}/.deps/${dep}-src" CACHE PATH "" FORCE)
  endif()
endforeach()
find_package(OpenCV 4 QUIET COMPONENTS core imgproc imgcodecs highgui videoio features2d calib3d ml objdetect dnn face photo ximgproc)
if(NOT OpenCV_FOUND)
  if(NOT CV40_FETCH_DEPS)
    message(FATAL_ERROR "Install OpenCV 4 with contrib or enable CV40_FETCH_DEPS")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  set(BUILD_LIST "core,imgproc,imgcodecs,highgui,videoio,features2d,calib3d,ml,objdetect,dnn,face,photo,ximgproc" CACHE STRING "" FORCE)
  foreach(flag BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps BUILD_JAVA BUILD_opencv_python2 BUILD_opencv_python3 WITH_IPP WITH_OPENCL WITH_CUDA WITH_FFMPEG WITH_GSTREAMER WITH_OPENEXR WITH_OBSENSOR)
    set(${flag} OFF CACHE BOOL "" FORCE)
  endforeach()
  FetchContent_Declare(opencv_contrib URL https://github.com/opencv/opencv_contrib/archive/refs/tags/4.13.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(opencv_contrib)
  set(OPENCV_EXTRA_MODULES_PATH "${opencv_contrib_SOURCE_DIR}/modules" CACHE PATH "" FORCE)
  FetchContent_Declare(opencv URL https://github.com/opencv/opencv/archive/refs/tags/4.13.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(opencv)
  set(OpenCV_LIBS opencv_core opencv_imgproc opencv_imgcodecs opencv_highgui opencv_videoio opencv_features2d opencv_calib3d opencv_ml opencv_objdetect opencv_dnn opencv_face opencv_photo opencv_ximgproc)
  set(OpenCV_INCLUDE_DIRS "${opencv_SOURCE_DIR}/include" "${opencv_BINARY_DIR}" "${CMAKE_BINARY_DIR}")
  file(GLOB module_headers "${opencv_SOURCE_DIR}/modules/*/include" "${opencv_contrib_SOURCE_DIR}/modules/*/include")
  list(APPEND OpenCV_INCLUDE_DIRS ${module_headers})
endif()
find_package(dlib QUIET)
if(NOT dlib_FOUND)
  if(NOT CV40_FETCH_DEPS)
    message(FATAL_ERROR "Install dlib or enable CV40_FETCH_DEPS")
  endif()
  set(DLIB_JPEG_SUPPORT OFF CACHE BOOL "" FORCE)
  set(DLIB_PNG_SUPPORT OFF CACHE BOOL "" FORCE)
  set(DLIB_GIF_SUPPORT OFF CACHE BOOL "" FORCE)
  set(DLIB_WEBP_SUPPORT OFF CACHE BOOL "" FORCE)
  set(DLIB_USE_CUDA OFF CACHE BOOL "" FORCE)
  set(DLIB_NO_GUI_SUPPORT ON CACHE BOOL "" FORCE)
  set(DLIB_USE_BLAS OFF CACHE BOOL "" FORCE)
  set(DLIB_USE_LAPACK OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(dlib URL https://github.com/davisking/dlib/archive/refs/tags/v20.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(dlib)
endif()
file(GLOB CV40_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(cv40 ${CV40_SOURCES})
set_target_properties(cv40 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_include_directories(cv40 PRIVATE include)
target_include_directories(cv40 SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
if(dlib_SOURCE_DIR)
  target_include_directories(cv40 SYSTEM PRIVATE "${dlib_SOURCE_DIR}")
endif()
target_link_libraries(cv40 PRIVATE ${OpenCV_LIBS} dlib::dlib)
if(WIN32)
  target_link_libraries(cv40 PRIVATE shell32)
  target_compile_definitions(cv40 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_compile_definitions(cv40 PRIVATE CV40_DEFAULT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/data")
if(MSVC)
  target_compile_options(cv40 PRIVATE /utf-8 /W4 /bigobj)
else()
  target_compile_options(cv40 PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME algorithms COMMAND cv40 --self-test)
add_test(NAME cli_help COMMAND cv40 --help)
add_test(NAME cli_list COMMAND cv40 --list)
add_test(NAME cli_negative_frames COMMAND cv40 --run 8.5 --headless --max-frames -1)
add_test(NAME cli_zero_frames COMMAND cv40 --run 8.5 --headless --max-frames 0)
add_test(NAME cli_partial_number COMMAND cv40 --run 3.4 --seed 42oops)
add_test(NAME cli_invalid_input COMMAND cv40 --run 2.1 --input 1,broken)
add_test(NAME cli_conflicting_actions COMMAND cv40 --all-safe --run 3.4)
set_tests_properties(cli_negative_frames cli_zero_frames cli_partial_number cli_invalid_input
  cli_conflicting_actions PROPERTIES WILL_FAIL TRUE)
